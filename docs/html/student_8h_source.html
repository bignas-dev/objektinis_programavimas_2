258inline float calculateAverage(const std::vector<int>& arr) {
259 if (arr.empty()) return 0.0f;
260 int sum = 0;
261 for (int val : arr) {
262 sum += val;
263 }
264 return static_cast<float>(sum) / static_cast<float>(arr.size());

272inline float calculateMedian(std::vector<int> arr) {
273 if (arr.empty()) return 0.0f;
274 std::sort(arr.begin(), arr.end());
275 size_t size = arr.size();
276 if (size % 2 == 0) {
277 return (arr[size/2 - 1] + arr[size/2]) / 2.0f;
278 } else {
279 return static_cast<float>(arr[size/2]);
280 }
281}

322template<typename Container>
323Container readFromFile(const std::string& filename) {
324 Container students;
325 std::ifstream file(filename);
326 if (!file.is_open()) {
327 throw std::runtime_error("Klaida: nepavyko atidaryti failo " + filename);
328 }
329
330 std::string line;
331 std::getline(file, line);
332
333 while (std::getline(file, line)) {
334 if (line.empty()) continue;
335
336 std::istringstream iss(line);
337 std::string vardas, pavarde;
338 if (!(iss >> vardas >> pavarde)) continue;
339
340 Studentas s(vardas, pavarde);
341 int grade;
342 for (int i = 0; i < 5; ++i) {
343 if (iss >> grade) {
344 s.addTarpRez(grade);
345 }
346 }
347 iss >> grade;
348 s.setEgzRez(grade);
349 students.push_back(s);
350 }
351
352 file.close();
353 return students;
354}
