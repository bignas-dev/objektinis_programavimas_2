35inline float calculateMedian(std::vector<int> arr) {
36 if (arr.empty()) return 0.0f;
37 std::sort(arr.begin(), arr.end());
38 size_t size = arr.size();
39 if (size % 2 == 0) {
40 return (arr[size/2 - 1] + arr[size/2]) / 2.0f;
41 } else {
42 return static_cast<float>(arr[size/2]);
